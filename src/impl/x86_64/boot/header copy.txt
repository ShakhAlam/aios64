section .multiboot_header
align 8
header_start:
    dd 0xe85250d6               ; Magic number: Multiboot2
    dd 0                        ; Architecture: i386
    dd header_end - header_start ; Total header length
    dd 0x100000000 - (0xe85250d6 + 0 + (header_end - header_start)) ; Checksum

    ; Framebuffer tag
    dd 4                        ; Tag type (Framebuffer)
    dd 32                       ; Size of this tag (32 bytes)
    dd 1920                     ; Framebuffer width
    dd 1080                     ; Framebuffer height
    dd 32                       ; Bits per pixel (BPP)
    dd 1                        ; Framebuffer type (RGB)

    db 8                        ; Red mask size
    db 16                       ; Red field position
    db 8                        ; Green mask size
    db 8                        ; Green field position
    db 8                        ; Blue mask size
    db 0                        ; Blue field position
    dw 0                        ; Reserved (align to 32 bytes)

    ; End tag
    dw 0
    dw 0
    dd 8

header_end:
